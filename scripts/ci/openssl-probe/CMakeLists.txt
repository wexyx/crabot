cmake_minimum_required(VERSION 3.20)
project(CrabotOpenSSLProbe LANGUAGES C)
find_package(OpenSSL 3 REQUIRED COMPONENTS SSL Crypto)
foreach(library OPENSSL_SSL_LIBRARY OPENSSL_CRYPTO_LIBRARY)
    if(NOT "${${library}}" MATCHES "\\.a$")
        message(FATAL_ERROR "${library} must be a static archive: ${${library}}")
    endif()
endforeach()
add_executable(crabot-openssl-probe main.c)
target_link_libraries(crabot-openssl-probe PRIVATE OpenSSL::SSL OpenSSL::Crypto)
